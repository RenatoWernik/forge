// a linha do check (ok ou FAIL) cujo nome começa pelo prefixo dado
export const linha = (r, nome) => r.linhas.find((l) => /^(ok {4}|FAIL {2})/.test(l) && l.slice(6).startsWith(nome));
const fim = (r) => '\n--- saída (fim) ---\n' + r.out.slice(-1800);
export const esperaFail = (r, nome, ...trechos) => {
  const l = linha(r, nome);
  assert.ok(l, `não há a linha «${nome}» no check` + fim(r));
  assert.ok(l.startsWith('FAIL  '), `a linha «${nome}» devia ser FAIL:\n${l}`);
  for (const t of trechos) assert.ok(typeof t === 'string' ? l.includes(t) : t.test(l), `o detalhe não nomeia «${t}»:\n${l}`);
  assert.equal(r.status, 1, 'um FAIL tem de sair com 1' + fim(r));
};
// FAIL numa das linhas learnings (C2 e C3 não dizem qual; sem git a linha do C4 fica ok, por isso o FAIL só pode vir da validação)
export const esperaFailLearnings = (r, ...trechos) => {
  const ls = r.linhas.filter((l) => /^FAIL {2}learnings: /.test(l));
  assert.ok(ls.length, 'nenhuma linha «FAIL  learnings: ...» no check' + fim(r));
  for (const t of trechos) assert.ok(ls.some((l) => (typeof t === 'string' ? l.includes(t) : t.test(l))), `nenhuma linha FAIL learnings nomeia «${t}»:\n${ls.join('\n')}`);
  assert.equal(r.status, 1, 'um FAIL tem de sair com 1' + fim(r));
};
export const esperaOk = (r, nome, exato) => {
  const l = linha(r, nome);
  assert.ok(l, `não há a linha «${nome}» no check` + fim(r));
  if (exato !== undefined) assert.equal(l, 'ok    ' + exato);
  else assert.ok(l.startsWith('ok    '), `a linha «${nome}» devia ser ok (alarme falso):\n${l}`);
  assert.deepEqual(r.linhas.filter((x) => /^FAIL {2}/.test(x)), [], 'há FAIL num caso válido' + fim(r));
  assert.equal(r.status, 0, fim(r));
  assert.match(r.out, /All ok\./);
};
